# 𑘫𑘿𑘨𑘲𑘦𑘽𑘝 𑘢𑘹𑘫𑘪𑘹 𑘟𑘦𑘿𑘝𑘨

## 𑘨𑘰𑘕𑘢𑘝𑘿𑘨 𑘢𑘿𑘨𑘞𑘦

𑘭𑘿𑘪𑘭𑘿𑘝𑘲 𑘫𑘿𑘨𑘲 𑘨𑘰𑘕𑘿𑘧𑘰𑘥𑘲𑘬𑘹𑘎 𑘫𑘎𑘹 𑘢𑘿𑘨𑘪𑘨𑘿𑘝𑘦𑘰𑘡𑘹 ।
𑘎𑘿𑘬𑘝𑘿𑘨𑘲𑘧𑘎𑘳𑘩𑘰𑘪𑘝𑘽𑘭 𑘫𑘿𑘨𑘲 𑘨𑘰𑘕𑘰 𑘫𑘲𑘪𑘔𑘝𑘿𑘨𑘢𑘝𑘲 𑘮𑘨𑘿𑘬𑘲𑘝 ॥ १ ॥

𑘢𑘳𑘜𑘹 𑘦𑘳𑘎𑘰𑘦𑘲 𑘢𑘝𑘿𑘨 𑘩𑘲𑘮𑘲𑘝𑘹 𑘭𑘲𑘟𑘿𑘠 𑘎𑘹𑘩𑘹 ।
𑘭𑘎𑘩 𑘐𑘳𑘜𑘰𑘩𑘽𑘎𑘴𑘝 𑘦𑘰𑘡𑘿𑘧𑘪𑘨 𑘭𑘡𑘿𑘦𑘰𑘡𑘲𑘝 ॥ २ ॥

𑘁𑘕𑘿𑘗𑘰 𑘎𑘹𑘩𑘲 𑘊𑘎𑘳𑘜 𑘭𑘨𑘿𑘪 𑘎𑘰𑘨𑘥𑘰𑘨𑘲 𑘩𑘻𑘎 ।
𑘟𑘹𑘪𑘭𑘿𑘞𑘰𑘡 𑘪 𑘤𑘿𑘨𑘰𑘮𑘿𑘦𑘜 𑘪𑘴𑘝𑘿𑘝𑘲 𑘨𑘎𑘿𑘬𑘜 𑘎𑘨𑘰𑘪𑘹 ॥ ३ ॥

𑘭𑘽𑘭𑘿𑘞𑘰𑘡 𑘨𑘎𑘿𑘬𑘜 𑘎𑘨𑘳𑘡 𑘠𑘨𑘿𑘦 𑘪𑘴𑘟𑘿𑘠𑘲 𑘎𑘨𑘰𑘪𑘲 ।
𑘢𑘿𑘨𑘕𑘹𑘭 𑘭𑘳𑘏 𑘭𑘽𑘝𑘻𑘬 𑘨𑘰𑘏𑘳𑘡 𑘎𑘰𑘨𑘿𑘧 𑘪𑘿𑘮𑘰𑘪𑘹 ॥ ४ ॥

𑘮𑘹 𑘢𑘝𑘿𑘨 𑘢𑘿𑘨𑘦𑘰𑘜 𑘦𑘰𑘡𑘳𑘡 𑘪𑘨𑘿𑘝𑘜𑘳𑘎 𑘙𑘹𑘪𑘰𑘪𑘲 ।
𑘦𑘨𑘿𑘧𑘰𑘟𑘹𑘓𑘹 𑘄𑘩𑘿𑘩𑘽𑘑𑘡 𑘎𑘟𑘲 𑘮𑘲 𑘡 𑘮𑘻𑘪𑘹 ॥ ५ ॥
